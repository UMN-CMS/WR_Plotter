  \begin{column}{0.52\textwidth}
    \textbf{\textcolor{green!50!black}{Done}}
    \begin{itemize}\setlength{\itemsep}{2pt}
      \item \textbf{Signal width (0--1):} per-$(m_{W_R},m_N)$ on-shell
            resolution measured; window-robust Gaussian-core $\sigma$
            (vs.\ RooKeysPdf FWHM) is the stable definition.
      \item \textbf{Parameterization (2):} $\sigma/m_{W_R}$ vs.\
            $x=m_N/m_{W_R}$ and $m_{W_R}$; compact \texttt{poly3+mass}
            (LOMO-CV $\sim\!2$--$3\%$).
      \item \textbf{Spectra (3):} SR $m_{\ell\ell jj}$ / $m_{\ell J}$ MC
            shapes $+$ DY-CR \& flavour-CR data/MC.
      \item \textbf{In-window background fit (4):} 6 analytic functions,
            recentered at the window peak $\mu$; \emph{4 checks} (valid min.,
            accurate covariance, no param.\ at limit, monotonic) $+$ physical
            \emph{slope $\le 0$} $\Rightarrow$ \texttt{expo} \& \texttt{powlaw}
            pass cleanly.
      \item \textbf{Spurious signal (5):} fake yield from an $S{+}B$ fit to a
            \emph{signal-free} template; pull acceptance bands.
      \item \textbf{Signal injection (6):} Asimov recovery bias \emph{and}
            Poisson toys for pull/coverage (coverage $\approx 1$ mid-mass).
    \end{itemize}
  \end{column}

  % ---------------------------------------------------------------- TO COME
  \begin{column}{0.48\textwidth}
    \textbf{\textcolor{red!60!black}{To come}}

      \item \textbf{Boosted:} extend spurious / injection beyond the

      \item \textbf{Smooth-truth template} to de-noise the spurious / injection

      \item \textbf{Realistic inputs:} matched eras $+$ DY $K$-factor $+$

      \item \textbf{Systematic envelope:} vary window width, fit range, function
            choice (discrete profiling) $\to$ background bias systematic.
      \item \textbf{\textcolor{blue!60!black}{Combine:}} build datacards and run
            \texttt{Combine} for the \emph{expected limits} (asymptotic
            $\mathrm{CL}_s$) --- the final deliverable.
    \end{itemize}
